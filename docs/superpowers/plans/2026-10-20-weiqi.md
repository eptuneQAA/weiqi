"""棋盘基础规则：落子、提子、气、禁入点。"""

EMPTY = 0
BLACK = 1
WHITE = 2

STAR_POINTS = {
    9: (2, 4, 6),
    13: (3, 6, 9),
    19: (3, 9, 15),
}


def opponent(color):
    return WHITE if color == BLACK else BLACK


def color_name(color):
    return "黑" if color == BLACK else "白"


class Board:
    def __init__(self, size=19):
        if size not in STAR_POINTS:
            raise ValueError("棋盘大小只支持 9、13、19")
        self.size = size
        self.grid = [[EMPTY for _ in range(size)] for _ in range(size)]
        self.captures = {BLACK: 0, WHITE: 0}

    def copy(self):
        other = Board.__new__(Board)
        other.size = self.size
        other.grid = [row[:] for row in self.grid]
        other.captures = dict(self.captures)
        return other

    def in_bounds(self, x, y):
        return 0 <= x < self.size and 0 <= y < self.size

    def get(self, x, y):
        return self.grid[y][x]

    def set(self, x, y, color):
        self.grid[y][x] = color

    def neighbors(self, x, y):
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nx, ny = x + dx, y + dy
            if self.in_bounds(nx, ny):
                yield nx, ny

    def group_and_liberties(self, x, y):
        color = self.grid[y][x]
        if color == EMPTY:
            return set(), set()
        stones = set()
        liberties = set()
        stack = [(x, y)]
        while stack:
            cx, cy = stack.pop()
            if (cx, cy) in stones:
                continue
            stones.add((cx, cy))
            for nx, ny in self.neighbors(cx, cy):
                value = self.grid[ny][nx]
                if value == EMPTY:
                    liberties.add((nx, ny))
                elif value == color and (nx, ny) not in stones:
                    stack.append((nx, ny))
        return stones, liberties

    def play(self, x, y, color):
        if not self.in_bounds(x, y):
            return False, "落子超出棋盘"
        if self.grid[y][x] != EMPTY:
            return False, "该位置已有棋子"
        before = [row[:] for row in self.grid]
        before_captures = dict(self.captures)
        self.grid[y][x] = color
        captured = 0
        for nx, ny in self.neighbors(x, y):
            if self.grid[ny][nx] == opponent(color):
                stones, liberties = self.group_and_liberties(nx, ny)
                if not liberties:
                    for sx, sy in stones:
                        self.grid[sy][sx] = EMPTY
                    captured += len(stones)
        self.captures[color] += captured
        own_stones, own_liberties = self.group_and_liberties(x, y)
        if not own_liberties:
            self.grid = before
            self.captures = before_captures
            return False, "禁止自杀"
        return True, ""

    def is_legal(self, x, y, color):
        if not self.in_bounds(x, y) or self.grid[y][x] != EMPTY:
            return False
        test = self.copy()
        ok, _ = test.play(x, y, color)
        return ok

    def position_key(self):
        return tuple(tuple(row) for row in self.grid)

    def star_points(self):
        coords = STAR_POINTS[self.size]
        return [(x, y) for y in coords for x in coords]
```

from enum import Enum

from .board import BLACK, EMPTY, WHITE, Board, opponent
from .scoring import auto_dead_groups, compute_score


class Phase(Enum):
    PLAYING = "playing"
    SCORING = "scoring"
    FINISHED = "finished"


class GameState:
    def __init__(self, size=19, komi=7.5, first_color=BLACK):
        self.size = size
        self.komi = komi
        self.first_color = first_color
        self.board = Board(size)
        self.turn = first_color
        self.phase = Phase.PLAYING
        self.history = []
        self.positions = {self.board.position_key()}
        self.consecutive_passes = 0
        self.dead_points = set()
        self.score = None
        self.winner = None
        self.result_reason = ""
        self.resigned_by = None

    @property
    def last_move(self):
        for _color, x, y in reversed(self.history):
            if x is not None:
                return (x, y)
        return None

    @property
    def move_count(self):
        return len(self.history)

    def is_legal(self, x, y, color=None):
        if self.phase != Phase.PLAYING:
            return False, "当前不能落子"
        color = self.turn if color is None else color
        if color != self.turn:
            return False, "还没轮到你"
        if not self.board.in_bounds(x, y):
            return False, "落子超出棋盘"
        if self.board.get(x, y) != EMPTY:
            return False, "该位置已有棋子"
        test = self.board.copy()
        ok, err = test.play(x, y, color)
        if not ok:
            return False, err
        if test.position_key() in self.positions:
            return False, "劫争：不能重复之前的局面"
        return True, ""

    def play(self, x, y):
        ok, err = self.is_legal(x, y)
        if not ok:
            return False, err
        test = self.board.copy()
        ok, err = test.play(x, y, self.turn)
        if not ok:
            return False, err
        self.board = test
        self.positions.add(test.position_key())
        self.history.append((self.turn, x, y))
        self.consecutive_passes = 0
        self.turn = opponent(self.turn)
        return True, ""

    def pass_turn(self):
        if self.phase != Phase.PLAYING:
            return False, "当前不能停一手"
        self.history.append((self.turn, None, None))
        self.consecutive_passes += 1
        if self.consecutive_passes >= 2:
            self.start_scoring()
        else:
            self.turn = opponent(self.turn)
        return True, ""

    def start_scoring(self):
        if self.phase != Phase.PLAYING:
            return
        self.phase = Phase.SCORING
        self.dead_points = auto_dead_groups(self.board)
        self.score = compute_score(self.board, self.dead_points, self.komi)

    def set_dead_points(self, points):
        if self.phase != Phase.SCORING:
            return False
        valid = set()
        for x, y in points:
            if self.board.in_bounds(x, y) and self.board.get(x, y) != EMPTY:
                valid.add((int(x), int(y)))
        self.dead_points = valid
        self.score = compute_score(self.board, self.dead_points, self.komi)
        return True

    def confirm_score(self):
        if self.phase != Phase.SCORING:
            return False
        if self.score is None:
            self.score = compute_score(self.board, self.dead_points, self.komi)
        self.phase = Phase.FINISHED
        self.winner = self.score["winner"]
        self.result_reason = "数子"
        return True

    def resign(self, color):
        if self.phase == Phase.FINISHED:
            return False
        self.phase = Phase.FINISHED
        self.winner = "white" if color == BLACK else "black"
        self.resigned_by = color
        self.result_reason = "认输"
        return True

    def undo(self, count=2):
        if self.phase != Phase.PLAYING:
            return False
        if count < 1 or count > len(self.history):
            return False
        records = self.history[:-count]
        fresh = GameState(self.size, self.komi, self.first_color)
        for _color, x, y in records:
            if x is None:
                fresh.pass_turn()
            else:
                ok, _ = fresh.play(x, y)
                if not ok:
                    return False
        self.board = fresh.board
        self.turn = fresh.turn
        self.phase = fresh.phase
        self.history = fresh.history
        self.positions = fresh.positions
        self.consecutive_passes = fresh.consecutive_passes
        self.dead_points = set()
        self.score = None
        self.winner = None
        self.result_reason = ""
        self.resigned_by = None
        return True

    def snapshot(self):
        moves = []
        for color, x, y in self.history:
            moves.append({"color": color, "x": x, "y": y})
        return {
            "size": self.size,
            "komi": self.komi,
            "first_color": self.first_color,
            "turn": self.turn,
            "phase": self.phase.value,
            "moves": moves,
            "captures": {
                "black": self.board.captures[BLACK],
                "white": self.board.captures[WHITE],
            },
            "dead_points": [[x, y] for x, y in sorted(self.dead_points)],
            "score": self.score,
            "winner": self.winner,
            "reason": self.result_reason,
            "resigned_by": self.resigned_by,
        }

    @classmethod
    def from_snapshot(cls, data):
        game = cls(int(data["size"]), float(data["komi"]), int(data["first_color"]))
        for move in data.get("moves", []):
            if move.get("x") is None:
                game.pass_turn()
            else:
                ok, err = game.play(int(move["x"]), int(move["y"]))
                if not ok:
                    raise ValueError("无法恢复棋局：" + err)
        if data.get("dead_points"):
            game.set_dead_points(data["dead_points"])
        if data.get("phase") == Phase.FINISHED.value:
            if data.get("resigned_by") is not None:
                game.resign(int(data["resigned_by"]))
            else:
                game.confirm_score()
        return game

from .board import BLACK, EMPTY, WHITE, opponent


def _cleaned_grid(board, dead_points):
    grid = [row[:] for row in board.grid]
    for x, y in dead_points:
        if board.in_bounds(x, y) and grid[y][x] != EMPTY:
            grid[y][x] = EMPTY
    return grid


def compute_ownership(board, dead_points):
    grid = _cleaned_grid(board, dead_points)
    size = board.size
    owner = [[EMPTY for _ in range(size)] for _ in range(size)]
    for y in range(size):
        for x in range(size):
            if grid[y][x] != EMPTY:
                owner[y][x] = grid[y][x]
    visited = [[False for _ in range(size)] for _ in range(size)]
    for y in range(size):
        for x in range(size):
            if grid[y][x] != EMPTY or visited[y][x]:
                continue
            stack = [(x, y)]
            region = []
            borders = set()
            visited[y][x] = True
            while stack:
                cx, cy = stack.pop()
                region.append((cx, cy))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if not board.in_bounds(nx, ny):
                        continue
                    value = grid[ny][nx]
                    if value == EMPTY:
                        if not visited[ny][nx]:
                            visited[ny][nx] = True
                            stack.append((nx, ny))
                    else:
                        borders.add(value)
            if borders == {BLACK}:
                for rx, ry in region:
                    owner[ry][rx] = BLACK
            elif borders == {WHITE}:
                for rx, ry in region:
                    owner[ry][rx] = WHITE
    return owner


def compute_score(board, dead_points, komi):
    grid = _cleaned_grid(board, dead_points)
    owner = compute_ownership(board, dead_points)
    black_stones = white_stones = 0
    black_territory = white_territory = 0
    for y in range(board.size):
        for x in range(board.size):
            value = grid[y][x]
            if value == BLACK:
                black_stones += 1
            elif value == WHITE:
                white_stones += 1
            elif owner[y][x] == BLACK:
                black_territory += 1
            elif owner[y][x] == WHITE:
                white_territory += 1
    black_area = black_stones + black_territory
    white_area = white_stones + white_territory
    black_total = float(black_area)
    white_total = float(white_area) + float(komi)
    difference = black_total - white_total
    if abs(difference) < 1e-9:
        winner = "draw"
        margin = 0.0
    elif difference > 0:
        winner = "black"
        margin = difference
    else:
        winner = "white"
        margin = -difference
    return {
        "black_stones": black_stones,
        "white_stones": white_stones,
        "black_territory": black_territory,
        "white_territory": white_territory,
        "black_area": black_area,
        "white_area": white_area,
        "komi": float(komi),
        "black_total": black_total,
        "white_total": white_total,
        "winner": winner,
        "margin": margin,
        "dead_count": len(dead_points),
    }


def auto_dead_groups(board):
    owner = compute_ownership(board, set())
    dead = set()
    visited = set()
    for y in range(board.size):
        for x in range(board.size):
            color = board.grid[y][x]
            if color == EMPTY or (x, y) in visited:
                continue
            stones, liberties = board.group_and_liberties(x, y)
            visited |= stones
            has_eye = any(owner[ly][lx] == color for lx, ly in liberties)
            all_opponent = bool(liberties) and all(
                owner[ly][lx] == opponent(color) for lx, ly in liberties
            )
            if not has_eye and all_opponent:
                dead |= stones
    return dead


def toggle_group_dead(board, dead_points, x, y):
    if not board.in_bounds(x, y) or board.get(x, y) == EMPTY:
        return set(dead_points)
    stones, _ = board.group_and_liberties(x, y)
    if stones <= dead_points:
        return set(dead_points) - stones
    return set(dead_points) | stones


def score_text(score):
    if score is None:
        return "对局结束"
    if score["winner"] == "draw":
        return "和棋"
    side = "黑" if score["winner"] == "black" else "白"
    return "{}胜 {:.1f} 点".format(side, score["margin"])

from weiqi.core.board import BLACK, WHITE, Board
from weiqi.core.scoring import (
    auto_dead_groups, compute_score, score_text, toggle_group_dead,
)


class ScoringTest(unittest.TestCase):
    def _walled_board(self):
        board = Board(9)
        for y in range(9):
            board.set(1, y, BLACK)
            board.set(7, y, WHITE)
        return board

    def test_area_score_with_walls(self):
        score = compute_score(self._walled_board(), set(), 7.5)
        self.assertEqual(score["black_stones"], 9)
        self.assertEqual(score["white_stones"], 9)
        self.assertEqual(score["black_territory"], 9)
        self.assertEqual(score["white_territory"], 9)
        self.assertEqual(score["black_area"], 18)
        self.assertEqual(score["white_area"], 18)
        self.assertEqual(score["white_total"], 25.5)
        self.assertEqual(score["winner"], "white")
        self.assertAlmostEqual(score["margin"], 7.5)

    def test_dead_stone_detection(self):
        board = self._walled_board()
        board.set(0, 0, WHITE)
        dead = auto_dead_groups(board)
        self.assertEqual(dead, {(0, 0)})
        alive = compute_score(board, set(), 7.5)
        killed = compute_score(board, dead, 7.5)
        self.assertEqual(alive["black_area"], 17)
        self.assertEqual(killed["black_area"], 18)
        self.assertEqual(killed["white_area"], 18)

    def test_toggle_group_dead(self):
        board = self._walled_board()
        board.set(0, 0, WHITE)

import math
import random
import time

from ..core.board import BLACK, EMPTY, WHITE, opponent
from ..core.game import GameState, Phase
from ..core.scoring import compute_ownership

DIFFICULTY = {
    "入门": {"time": 0.3, "top": 5, "playout": 0},
    "普通": {"time": 1.0, "top": 10, "playout": 40},
    "较强": {"time": 2.5, "top": 14, "playout": 60},
}


class AIEngine:
    def __init__(self, difficulty="普通", seed=None):
        if difficulty not in DIFFICULTY:
            raise ValueError("未知难度：" + str(difficulty))
        self.difficulty = difficulty
        self.config = DIFFICULTY[difficulty]
        self.random = random.Random(seed)

    def choose_move(self, game, cancel_check=None):
        if game.phase != Phase.PLAYING:
            return None
        if cancel_check is not None and cancel_check():
            return None
        color = game.turn
        moves = self._candidates(game, color)
        if not moves:
            return None
        ranked = self._rank_moves(game, moves, color)
        if not ranked:
            return None
        if self.config["playout"] <= 0:
            best_score = ranked[0][1]
            pool = [m for m, s in ranked if best_score - s <= 1.5]
            return self.random.choice(pool or [ranked[0][0]])
        return self._monte_carlo(game, ranked, color, cancel_check)

    def _candidates(self, game, color):
        board = game.board
        size = board.size
        stones = [
            (x, y)
            for y in range(size)
            for x in range(size)
            if board.grid[y][x] != EMPTY
        ]
        candidates = set()
        if not stones:
            candidates.update(board.star_points())
        else:
            for sx, sy in stones:
                for dy in range(-2, 3):
                    for dx in range(-2, 3):
                        x, y = sx + dx, sy + dy
                        if board.in_bounds(x, y) and board.grid[y][x] == EMPTY:
                            candidates.add((x, y))
        extra = board.star_points() if len(stones) < 6 else []
        candidates.update(extra)
        result = []
        for x, y in candidates:
            ok, _ = game.is_legal(x, y, color)
            if ok and not self._is_own_eye(board, x, y, color):
                result.append((x, y))
        self.random.shuffle(result)
        return result

    def _is_own_eye(self, board, x, y, color):
        for nx, ny in board.neighbors(x, y):
            if board.grid[ny][nx] != color:
                return False
        bad_diagonals = 0
        for dx, dy in ((1, 1), (1, -1), (-1, 1), (-1, -1)):
            nx, ny = x + dx, y + dy
            if board.in_bounds(nx, ny) and board.grid[ny][nx] != color:
                bad_diagonals += 1
        return bad_diagonals <= 1

    def _rank_moves(self, game, moves, color):
        ranked = []
        for x, y in moves:
            score = self._heuristic(game, x, y, color)
            ranked.append(((x, y), score))
        ranked.sort(key=lambda item: item[1], reverse=True)
        return ranked[: max(1, self.config["top"])]

    def _heuristic(self, game, x, y, color):
        board = game.board
        trial = board.copy()
        ok, _ = trial.play(x, y, color)
        if not ok:
            return -9999.0
        captured = trial.captures[color] - board.captures[color]
        own_stones, own_liberties = trial.group_and_liberties(x, y)
        score = captured * 120.0 + min(len(own_liberties), 5) * 6.0
        score += len(own_stones) * 0.4
        for nx, ny in trial.neighbors(x, y):
            if trial.grid[ny][nx] == color:
                score += 1.5
            elif trial.grid[ny][nx] == opponent(color):
                stones, liberties = trial.group_and_liberties(nx, ny)
                if len(liberties) == 1:
                    score += 8.0
                elif len(liberties) == 2:
                    score += 2.0
                score += min(len(stones), 4) * 0.3
        last = game.last_move
        if last is not None:
            distance = max(abs(last[0] - x), abs(last[1] - y))
            score += max(0.0, 4.0 - distance)
        if (x == 0 or y == 0 or x == board.size - 1 or y == board.size - 1):
            if game.move_count < board.size * 2:
                score -= 6.0
        score += self.random.random() * 0.5
        return score

    def _monte_carlo(self, game, ranked, color, cancel_check):
        deadline = time.monotonic() + self.config["time"]
        stats = {move: [0, 0.0] for move, _score in ranked}
        total = 0
        while time.monotonic() < deadline:
            if cancel_check is not None and cancel_check():
                return None
            move = self._select(stats, total)
            stats[move][0] += 1
            total += 1
            reward = self._playout_value(game, move, color, deadline, cancel_check)
            if reward is None:
                return None
            stats[move][1] += reward
        if total == 0:
            return ranked[0][0]
        return max(
            stats,
            key=lambda m: (stats[m][0], stats[m][1] / max(stats[m][0], 1)),
        )

    def _select(self, stats, total):
        untried = [m for m, value in stats.items() if value[0] == 0]
        if untried:
            return self.random.choice(untried)
        log_total = math.log(max(total, 2))
        best_move = None
        best_value = -1.0
        for move, (visits, wins) in stats.items():
            value = wins / visits + 0.8 * math.sqrt(log_total / visits)
            if value > best_value:
                best_value = value
                best_move = move
        return best_move

    def _playout_value(self, game, move, color, deadline, cancel_check):
        board = game.board.copy()
        ok, _ = board.play(move[0], move[1], color)
        if not ok:
            return 0.0

        max_moves = 120 if board.size <= 9 else 80
        for _ in range(max_moves):
            if time.monotonic() >= deadline:
                break
            if cancel_check is not None and cancel_check():
                return None
            options = self._local_empty_points(board)
            if not options:
                passes += 1
                if passes >= 2:
                    break
                turn = opponent(turn)
                continue
            self.random.shuffle(options)
            played = False
            for x, y in options[:20]:
                if self._is_own_eye(board, x, y, turn):
                    continue
                moved, _ = board.play(x, y, turn)

                    played = True
                    passes = 0
                    turn = opponent(turn)
                    break
            if not played:
                passes += 1
                if passes >= 2:
                    break
                turn = opponent(turn)
        owner = compute_ownership(board, set())
        my_area = 0
        other_area = 0
        for y in range(board.size):
            for x in range(board.size):
                value = owner[y][x]
                if value == color:
                    my_area += 1
                elif value == opponent(color):

        threshold = other_area + (game.komi if color == BLACK else 0.0)

                            break

"""JSON 行协议与输入校验。"""

import json

PROTOCOL_VERSION = 1
DISCOVERY_MAGIC = "WEIQI_DISCOVERY"
DISCOVERY_PORT = 45678
DEFAULT_TCP_PORT = 45679
MAX_CHAT_LENGTH = 500
MAX_NAME_LENGTH = 20


class ProtocolError(ValueError):
    pass


def encode_message(message):
    text = json.dumps(message, ensure_ascii=False, separators=(",", ":"))
    return (text + "\n").encode("utf-8")


def decode_message(line):
    if isinstance(line, bytes):
        try:
            text = line.decode("utf-8")
        except UnicodeDecodeError:
            raise ProtocolError("消息不是 UTF-8 编码")
    else:
        text = line
    text = text.strip()
    if not text:
        raise ProtocolError("空消息")
    try:
        data = json.loads(text)
    except ValueError:
        raise ProtocolError("JSON 格式错误")
    if not isinstance(data, dict):
        raise ProtocolError("消息必须是对象")
    if not isinstance(data.get("type"), str):
        raise ProtocolError("缺少消息类型")
    return data


def clean_name(name):
    if not isinstance(name, str):
        raise ProtocolError("名称格式错误")
    value = name.strip()
    if not 1 <= len(value) <= MAX_NAME_LENGTH:
        raise ProtocolError("名称长度需为 1-20 字符")
    if any(ord(ch) < 32 for ch in value):
        raise ProtocolError("名称包含控制字符")
    return value


def clean_chat_text(text):
    if not isinstance(text, str):
        raise ProtocolError("聊天内容格式错误")
    value = text.strip()
    if not value:
        raise ProtocolError("聊天内容为空")
    if len(value) > MAX_CHAT_LENGTH:
        raise ProtocolError("聊天内容不能超过 500 字符")
    if any(ord(ch) < 32 for ch in value):
        raise ProtocolError("聊天内容包含控制字符")
    return value

"""UDP 房间广播与发现。"""

import json
import time

from PyQt5.QtCore import QObject, QTimer, pyqtSignal
from PyQt5.QtNetwork import QAbstractSocket, QHostAddress, QUdpSocket

from .protocol import DISCOVERY_MAGIC, DISCOVERY_PORT, PROTOCOL_VERSION


class RoomBroadcaster(QObject):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.socket = QUdpSocket(self)
        self.timer = QTimer(self)
        self.timer.timeout.connect(self._broadcast)
        self.payload = None

    def start(self, room, tcp_port, board_size, host_color):
        self.payload = {
            "magic": DISCOVERY_MAGIC,
            "version": PROTOCOL_VERSION,
            "room": room,
            "tcp_port": int(tcp_port),
            "board_size": int(board_size),
            "host_color": int(host_color),
        }
        self._broadcast()
        self.timer.start(2000)

    def _broadcast(self):
        if not self.payload:
            return
        data = json.dumps(self.payload, ensure_ascii=False).encode("utf-8")
        self.socket.writeDatagram(
            data, QHostAddress(QHostAddress.Broadcast), DISCOVERY_PORT
        )

    def stop(self):
        self.timer.stop()
        self.payload = None
        self.socket.close()


class RoomListener(QObject):
    rooms_changed = pyqtSignal(list)

    def __init__(self, parent=None):
        super().__init__(parent)
        self.socket = QUdpSocket(self)
        self.socket.readyRead.connect(self._read_pending)
        self.timer = QTimer(self)
        self.timer.timeout.connect(self._cleanup)
        self.rooms = {}

    def start(self):
        flags = (
            QAbstractSocket.ShareAddress | QAbstractSocket.ReuseAddressHint
        )
        ok = self.socket.bind(QHostAddress.AnyIPv4, DISCOVERY_PORT, flags)
        if ok:
            self.timer.start(2000)
        return ok

    def stop(self):
        self.timer.stop()
        self.socket.close()
        self.rooms.clear()

    def current_rooms(self):
        return self._sorted_rooms()

    def _read_pending(self):
        now = time.monotonic()
        while self.socket.hasPendingDatagrams():
            size = self.socket.pendingDatagramSize()
            data, host, _port = self.socket.readDatagram(size)
            try:
                info = json.loads(bytes(data).decode("utf-8"))
            except (ValueError, UnicodeDecodeError):
                continue
            if not isinstance(info, dict):
                continue
            if info.get("magic") != DISCOVERY_MAGIC:
                continue
            if info.get("version") != PROTOCOL_VERSION:
                continue
            tcp_port = info.get("tcp_port")
            room_name = info.get("room")
            board_size = info.get("board_size")
            if not isinstance(tcp_port, int) or not isinstance(room_name, str):
                continue
            if board_size not in (9, 13, 19):
                continue
            ip = host.toString()
            self.rooms[(ip, tcp_port)] = {
                "room": room_name,
                "ip": ip,
                "port": tcp_port,
                "board_size": board_size,
                "host_color": info.get("host_color", 1),
                "seen": now,
            }
        self.rooms_changed.emit(self._sorted_rooms())

    def _cleanup(self):
        now = time.monotonic()
        expired = [
            key for key, value in self.rooms.items()
            if now - value["seen"] > 5.0
        ]
        for key in expired:
            del self.rooms[key]
        if expired:
            self.rooms_changed.emit(self._sorted_rooms())

    def _sorted_rooms(self):
        values = list(self.rooms.values())
        values.sort(key=lambda item: (item["room"], item["ip"], item["port"]))
        return values
```

        self.heartbeat.start(5000)

    def start(self):
        ok = self.server.listen(QHostAddress.AnyIPv4, self.port)
        if not ok:
            return False, self.server.errorString()
        return True, ""

    def stop(self):
        self.closing = True
        self.heartbeat.stop()
        if self.socket is not None:
            try:
                self.socket.write(encode_message(
                    {"type": "bye", "reason": "主机已关闭房间"}
                ))
                self.socket.flush()
            except Exception:
                pass
            self.socket.disconnectFromHost()
            self.socket.deleteLater()
            self.socket = None
        self.client_ready = False
        self.server.close()

    def _on_new_connection(self):
        while self.server.hasPendingConnections():
            socket = self.server.nextPendingConnection()
            if self.socket is not None:
                try:
                    socket.write(encode_message(
                        {"type": "bye", "reason": "房间已有玩家"}
                    ))
                    socket.flush()
                except Exception:
                    pass
                socket.disconnectFromHost()
                socket.deleteLater()
                continue
            self.socket = socket
            self.buffer = b""
            self.client_ready = False
            self.last_seen = time.monotonic()
            socket.readyRead.connect(self._on_ready_read)
            socket.disconnected.connect(self._on_disconnected)

    def _on_ready_read(self):
        if self.socket is None:
            return
        self.buffer += bytes(self.socket.readAll())
        while b"\n" in self.buffer:
            line, self.buffer = self.buffer.split(b"\n", 1)
            if not line.strip():
                continue
            try:
                message = decode_message(line)
            except ProtocolError as exc:
                self._protocol_error(str(exc))
                continue
            self.last_seen = time.monotonic()
            self.error_count = 0
            self._handle(message)

    def _handle(self, message):
        message_type = message["type"]
        if not self.client_ready:
            if message_type != "hello":
                self._send_error("请先发送 hello")
                return
            try:
                version = int(message.get("version", 0))
                name = clean_name(message.get("name", ""))
            except (TypeError, ValueError, ProtocolError) as exc:
                self._send_error(str(exc))
                return
            if version != PROTOCOL_VERSION:
                self._send_error("协议版本不一致，请使用相同版本")
                self._disconnect_client("版本不一致")
                return
            self.guest_name = name
            self.client_ready = True
            state = self.game.snapshot()
            self._send({
                "type": "welcome",
                "version": PROTOCOL_VERSION,
                "room": self.room_name,
                "board_size": self.game.size,
                "komi": self.game.komi,
                "host_color": self.host_color,
                "guest_color": self.guest_color,
                "state": state,
            })
            self.client_joined.emit({"name": name, "state": state})
            return

        if message_type == "ping":
            self._send({"type": "pong"})
        elif message_type == "pong":
            pass
        elif message_type == "chat":
            try:
                text = clean_chat_text(message.get("text", ""))
            except ProtocolError as exc:
                self._send_error(str(exc))
                return
            self._broadcast_chat(self.guest_name, text)
        elif message_type == "move":
            self._guest_move(message)
        elif message_type == "pass":
            self._guest_pass()
        elif message_type == "resign":
            if self.game.phase == Phase.FINISHED:
                self._send_error("对局已结束")
                return
            self.game.resign(self.guest_color)
            self._after_change()
        elif message_type in ("mark_dead", "confirm_score"):
            self._send_error("只有主机可以操作数子")
        elif message_type == "leave":
            self._send({"type": "bye", "reason": "客机已离开"})
            self._disconnect_client("客机已离开", notify=True)
        else:
            self._send_error("未知消息类型")

    def _guest_move(self, message):
        if self.game.phase != Phase.PLAYING:
            self._send_move_result(False, "当前不能落子")
            return
        if self.game.turn != self.guest_color:
            self._send_move_result(False, "还没轮到你")
            return
        try:
            x = int(message.get("x"))
            y = int(message.get("y"))
        except (TypeError, ValueError):
            self._send_move_result(False, "坐标格式错误")
            return
        ok, err = self.game.play(x, y)
        self._send_move_result(ok, err)
        if ok:
            self._after_change()

    def _guest_pass(self):
        if self.game.phase != Phase.PLAYING:
            self._send_state()
            return
        if self.game.turn != self.guest_color:
            self._send_error("还没轮到你")
            return
        self.game.pass_turn()
        self._after_change()

    def host_play(self, x, y):
        if not self.client_ready:
            return False, "对手尚未加入"
        if self.game.phase != Phase.PLAYING:
            return False, "当前不能落子"
        if self.game.turn != self.host_color:
            return False, "还没轮到你"
        ok, err = self.game.play(x, y)
        if ok:
            self._after_change()
        return ok, err

    def host_pass(self):
        if not self.client_ready:
            return False, "对手尚未加入"
        if self.game.phase != Phase.PLAYING:
            return False, "当前不能停一手"
        if self.game.turn != self.host_color:
            return False, "还没轮到你"
        self.game.pass_turn()
        self._after_change()
        return True, ""

    def host_resign(self):
        if not self.client_ready:
            return False, "对手尚未加入"
        if self.game.phase == Phase.FINISHED:
            return False, "对局已结束"
        self.game.resign(self.host_color)
        self._after_change()
        return True, ""

    def host_mark_dead(self, points):
        if not self.client_ready:
            return False, "对手尚未加入"
        if self.game.phase != Phase.SCORING:
            return False, "当前不在数子阶段"
        self.game.set_dead_points(points)
        self._after_change()
        return True, ""

    def host_confirm_score(self):
        if not self.client_ready:
            return False, "对手尚未加入"
        if self.game.phase != Phase.SCORING:
            return False, "当前不在数子阶段"
        self.game.confirm_score()
        self._after_change()
        return True, ""

    def host_chat(self, text):
        try:
            clean = clean_chat_text(text)
        except ProtocolError as exc:
            return False, str(exc)
        self._broadcast_chat(self.host_name, clean)
        return True, ""

    def _broadcast_chat(self, name, text):
        message = {
            "type": "chat",
            "name": name,
            "text": text,
            "timestamp": int(time.time() * 1000),
        }
        self._send(message)
        self.chat_received.emit(message)

    def _after_change(self):
        state = self.game.snapshot()
        self._send({"type": "state", "state": state})
        self.state_changed.emit(state)
        if self.game.phase == Phase.FINISHED:
            self.game_finished.emit(self._result())

    def _result(self):
        return {
            "winner": self.game.winner,
            "reason": self.game.result_reason,
            "score": self.game.score,
            "resigned_by": self.game.resigned_by,
            "state": self.game.snapshot(),
        }

    def _send_move_result(self, ok, error):
        self._send({
            "type": "move_result",
            "ok": bool(ok),
            "error": error,
            "state": self.game.snapshot(),
        })
        if ok:
            self.state_changed.emit(self.game.snapshot())

    def _send_state(self):
        self._send({"type": "state", "state": self.game.snapshot()})

    def _send_error(self, text):
        self._send({"type": "error", "message": text})
        self.error_count += 1
        if self.error_count > 5:
            self._disconnect_client("非法消息过多", notify=True)

    def _protocol_error(self, text):
        self._send_error(text)

    def _send(self, message):
        if self.socket is None:
            return
        try:
            self.socket.write(encode_message(message))
            self.socket.flush()
        except Exception:
            pass

    def _heartbeat(self):
        if self.socket is None or not self.client_ready:
            return
        if time.monotonic() - self.last_seen > 15.0:
            self._disconnect_client("连接超时", notify=True)
            return
        self._send({"type": "ping"})

    def _disconnect_client(self, reason, notify=False):
        if self.socket is None:
            return
        try:
            self.socket.write(encode_message({"type": "bye", "reason": reason}))
            self.socket.flush()
            self.socket.disconnectFromHost()
        except Exception:
            pass
        self.socket.deleteLater()
        self.socket = None
        was_ready = self.client_ready
        self.client_ready = False
        self.guest_name = ""
        self.error_count = 0
        if notify and was_ready:
            self.client_left.emit(reason)

    def _on_disconnected(self):
        if self.socket is None:
            return
        self.socket.deleteLater()
        self.socket = None
        was_ready = self.client_ready
        self.client_ready = False
        self.guest_name = ""
        if was_ready and not self.closing:
            self.client_left.emit("对手已断开")
```

"""客机端 TCP 连接、心跳与消息分发。"""

import time

from PyQt5.QtCore import QObject, QTimer, pyqtSignal
from PyQt5.QtNetwork import QAbstractSocket, QTcpSocket

from .protocol import (
    PROTOCOL_VERSION, ProtocolError, clean_name, decode_message,
    encode_message,
)


class GameClient(QObject):
    welcome_received = pyqtSignal(dict)
    state_changed = pyqtSignal(dict)
    game_finished = pyqtSignal(dict)
    chat_received = pyqtSignal(dict)
    disconnected = pyqtSignal(str)
    error_occurred = pyqtSignal(str)

    def __init__(self, parent=None):
        super().__init__(parent)
        self.socket = QTcpSocket(self)
        self.socket.connected.connect(self._on_connected)
        self.socket.readyRead.connect(self._on_ready_read)
        self.socket.disconnected.connect(self._on_disconnected)
        self.socket.error.connect(self._on_socket_error)
        self.buffer = b""
        self.name = ""
        self.host = ""
        self.port = 0
        self.last_seen = time.monotonic()
        self.notified = False
        self.closing = False
        self.heartbeat = QTimer(self)
        self.heartbeat.timeout.connect(self._heartbeat)

    def connect_to_host(self, host, port, name):
        self.host = host
        self.port = int(port)
        self.name = clean_name(name)
        self.buffer = b""
        self.notified = False
        self.closing = False
        self.socket.connectToHost(host, self.port)

    def _on_connected(self):
        self.last_seen = time.monotonic()
        self.notified = False
        self.heartbeat.start(5000)
        self._send({
            "type": "hello",
            "version": PROTOCOL_VERSION,
            "name": self.name,
        })

    def _on_ready_read(self):
        self.buffer += bytes(self.socket.readAll())
        while b"\n" in self.buffer:
            line, self.buffer = self.buffer.split(b"\n", 1)
            if not line.strip():
                continue
            try:
                message = decode_message(line)
            except ProtocolError as exc:
                self.error_occurred.emit(str(exc))
                continue
            self.last_seen = time.monotonic()
            self._handle(message)

    def _handle(self, message):
        message_type = message.get("type")
        if message_type == "welcome":
            if int(message.get("version", 0)) != PROTOCOL_VERSION:
                self._fail("协议版本不一致")
                return
            self.welcome_received.emit(message)
        elif message_type == "state":
            state = message.get("state")
            if isinstance(state, dict):
                self.state_changed.emit(state)
        elif message_type == "move_result":
            if message.get("ok") and isinstance(message.get("state"), dict):
                self.state_changed.emit(message["state"])
            else:
                self.error_occurred.emit(message.get("error", "落子失败"))
        elif message_type == "result":
            self.game_finished.emit(message)
        elif message_type == "chat":
            self.chat_received.emit(message)
        elif message_type == "error":
            self.error_occurred.emit(message.get("message", "未知错误"))
        elif message_type == "ping":
            self._send({"type": "pong"})
        elif message_type == "pong":
            pass
        elif message_type == "bye":
            self._fail(message.get("reason", "主机已关闭房间"))

    def send_move(self, x, y):
        self._send({"type": "move", "x": int(x), "y": int(y)})

    def send_pass(self):
        self._send({"type": "pass"})

    def send_resign(self):
        self._send({"type": "resign"})

    def send_chat(self, text):
        self._send({"type": "chat", "text": text})

    def _send(self, message):
        if self.socket.state() != QAbstractSocket.ConnectedState:
            return
        try:
            self.socket.write(encode_message(message))
            self.socket.flush()
        except Exception:
            pass

    def _heartbeat(self):
        if self.socket.state() != QAbstractSocket.ConnectedState:
            return
        if time.monotonic() - self.last_seen > 15.0:
            self._fail("连接超时")
            return
        self._send({"type": "ping"})

    def _on_socket_error(self, _error):
        if not self.closing:
            self._fail(self.socket.errorString())

    def _on_disconnected(self):
        self.heartbeat.stop()
        if not self.notified and not self.closing:
            self._fail("连接已断开")

    def _fail(self, reason):
        if self.notified:
            return
        self.notified = True
        self.heartbeat.stop()
        self.disconnected.emit(reason)
        self.close()

    def close(self):
        self.closing = True
        self.heartbeat.stop()
        if self.socket.state() != QAbstractSocket.UnconnectedState:
            self.socket.disconnectFromHost()
        self.socket.close()

from ..core.board import BLACK, EMPTY, WHITE
from . import theme

LETTERS = "ABCDEFGHJKLMNOPQRST"


class BoardWidget(QWidget):
    point_clicked = pyqtSignal(int, int)

    def __init__(self, parent=None):
        super().__init__(parent)
        self.board = None
        self.size = 19
        self.last_move = None
        self.interactive = False
        self.marking = False
        self.dead_points = set()
        self.hover = None
        self.hover_color = BLACK
        self.setMinimumSize(460, 460)
        self.setMouseTracking(True)

    def set_state(
        self, board, last_move=None, interactive=False, marking=False,
        dead_points=None, hover_color=BLACK,
    ):
        self.board = board
        self.size = board.size
        self.last_move = last_move

        self.marking = marking
        self.dead_points = set(dead_points or set())
        self.hover_color = hover_color

    def _metrics(self):
        side = min(self.width(), self.height())
        margin = max(30.0, side * 0.075)
        length = max(1.0, side - 2.0 * margin)
        cell = length / max(1, self.size - 1)
        left = (self.width() - side) / 2.0 + margin
        top = (self.height() - side) / 2.0 + margin
        return left, top, cell, side

    def _pixel_to_point(self, pos):
        left, top, cell, _side = self._metrics()
        x = int(round((pos.x() - left) / cell))
        y = int(round((pos.y() - top) / cell))
        if not (0 <= x < self.size and 0 <= y < self.size):
            return None
        cx = left + x * cell
        cy = top + y * cell
        if (pos.x() - cx) ** 2 + (pos.y() - cy) ** 2 <= (cell * 0.48) ** 2:
            return x, y
        return None

    def paintEvent(self, _event):
        if self.board is None:
            return
        painter = QPainter(self)
        painter.setRenderHint(QPainter.Antialiasing, True)
        painter.fillRect(self.rect(), theme.BACKGROUND)
        left, top, cell, _side = self._metrics()

        grid_pen = QPen(theme.GRID)

        painter.setPen(grid_pen)
        for index in range(self.size):
            px = left + index * cell
            py = top + index * cell
            painter.drawLine(QPointF(px, top), QPointF(px, top + (self.size - 1) * cell))
            painter.drawLine(QPointF(left, py), QPointF(left + (self.size - 1) * cell, py))

    def _draw_stone(self, painter, cx, cy, cell, color, x, y):

            offset = radius * 0.55
            painter.drawLine(
                QPointF(cx - offset, cy - offset), QPointF(cx + offset, cy + offset)

            self.hover = None
            self.update()

    def mousePressEvent(self, event):
        if event.button() != Qt.LeftButton:
            return
        point = self._pixel_to_point(event.pos())
        if point is not None:

    move_ready = pyqtSignal(object)

        self.engine = engine

        self.chat.set_enabled(False)
        self.chat.add_system(reason)
        self.statusBar().showMessage(reason)
        self._update_ui()
        if not was_connected:
            QMessageBox.warning(self, "连接失败", reason)

    def _apply_snapshot(self, data):
        if not isinstance(data, dict):
            return
        try:
            self.game = GameState.from_snapshot(data)
        except (KeyError, TypeError, ValueError) as exc:
            self._show_error("同步棋局失败：" + str(exc))
            return
        self._update_ui()
        if self.game.phase == Phase.FINISHED and not self.result_shown:
            self._show_result_from_game()

    def _on_remote_finished(self, message):
        if self.result_shown:
            return
        self._show_result(
            message.get("score"),
            message.get("reason", "") or self.game.result_reason,
        )

    def _is_connected(self):
        if self.mode == MODE_HOST:
            return self.guest_joined
        if self.mode == MODE_GUEST:
            return self.client_connected
        return self.mode == MODE_AI

    def _is_local_turn(self):
        if self.game.phase != Phase.PLAYING:
            return False
        if self.mode == MODE_AI:
            return self.game.turn == self.human_color
        if self.mode == MODE_HOST:
            return self.guest_joined and self.game.turn == self.human_color
        if self.mode == MODE_GUEST:
            return self.client_connected and self.game.turn == self.human_color
        return False

    def _can_mark_dead(self):
        if self.game.phase != Phase.SCORING:
            return False
        if self.mode == MODE_AI:
            return True
        return self.mode == MODE_HOST and self.guest_joined

    def _can_resign(self):
        if self.game.phase not in (Phase.PLAYING, Phase.SCORING):
            return False
        return self._is_connected()

    def _can_undo(self):
        return (
            self.mode == MODE_AI
            and self.game.phase == Phase.PLAYING
            and bool(self.game.history)
            and (self.game.turn == self.human_color or self.ai_worker is not None)
        )

    def _chat_available(self):
        if self.mode == MODE_HOST:
            return self.guest_joined
        if self.mode == MODE_GUEST:
            return self.client_connected
        return False

    def _update_ui(self):
        phase = self.game.phase
        interactive = phase == Phase.PLAYING and self._is_local_turn()
        marking = phase == Phase.SCORING and self._can_mark_dead()
        self.board_widget.set_state(
            self.game.board, self.game.last_move, interactive, marking,
            self.game.dead_points, self.human_color,
        )
        mode_names = {
            MODE_IDLE: "未开始", MODE_AI: "人机对战",
            MODE_HOST: "主机", MODE_GUEST: "客机",
        }
        self.mode_label.setText("模式：{}".format(mode_names.get(self.mode, "未开始")))
        if phase == Phase.PLAYING:
            if self.mode in (MODE_HOST, MODE_GUEST) and not self._is_connected():
                self.turn_label.setText("等待对手…")
            else:
                self.turn_label.setText("轮到：{}方".format(color_name(self.game.turn)))
        elif phase == Phase.SCORING:
            self.turn_label.setText("数子阶段")
        else:
            self.turn_label.setText("对局结束")
        captures = self.game.board.captures
        self.capture_label.setText(
            "提子：黑 {} / 白 {}".format(captures[BLACK], captures[WHITE])
        )
        self.move_label.setText("手数：{}".format(self.game.move_count))
        if self.mode == MODE_HOST and self.server is not None:
            self.room_label.setText(
                "房间：{}  端口：{}".format(
                    self.server.room_name, self.server.port
                )
            )
        elif self.mode == MODE_GUEST:
            self.room_label.setText("客机模式")
        else:
            self.room_label.setText("")
        if phase == Phase.SCORING and self.game.score:
            score = self.game.score
            self.score_label.setText(
                "黑 {} 点 / 白 {} + {:.1f} = {:.1f} 点".format(
                    score["black_area"], score["white_area"],
                    score["komi"], score["white_total"],
                )
            )
        else:
            self.score_label.setText("")
        self.pass_button.setEnabled(interactive)
        self.resign_button.setEnabled(self._can_resign())
        self.undo_button.setEnabled(self._can_undo())
        self.confirm_button.setEnabled(marking)
        self.chat.set_enabled(self._chat_available())

    def _on_point_clicked(self, x, y):
        if self.game.phase == Phase.SCORING:
            if not self._can_mark_dead():
                return
            if self.game.board.get(x, y) == EMPTY:
                return
            new_dead = toggle_group_dead(
                self.game.board, self.game.dead_points, x, y
            )
            if self.mode == MODE_HOST and self.server is not None:
                ok, err = self.server.host_mark_dead(sorted(new_dead))
                if not ok:
                    self.statusBar().showMessage(err, 3000)
            else:
                self.game.set_dead_points(sorted(new_dead))
                self._update_ui()
                if self.game.score:
                    self.statusBar().showMessage(
                        score_text(self.game.score), 5000
                    )
            return
        if self.game.phase != Phase.PLAYING:
            return
        if not self._is_local_turn():
            self.statusBar().showMessage("还没轮到你", 3000)
            return
        if self.mode == MODE_AI:
            ok, err = self.game.play(x, y)
            if not ok:
                self.statusBar().showMessage(err, 3000)
                return
            self._update_ui()
            if self.game.phase == Phase.SCORING:
                self.statusBar().showMessage("双方停一手，进入数子", 5000)
            else:
                self._start_ai()
        elif self.mode == MODE_HOST:
            if self.server is None:
                return
            ok, err = self.server.host_play(x, y)
            if not ok:
                self.statusBar().showMessage(err, 3000)
        elif self.mode == MODE_GUEST:
            if self.client is None:
                return
            ok, err = self.game.is_legal(x, y)
            if not ok:
                self.statusBar().showMessage(err, 3000)
                return
            self.client.send_move(x, y)
            self.statusBar().showMessage("已发送落子", 1500)

    def _pass(self):
        if self.game.phase != Phase.PLAYING:
            return
        if self.mode == MODE_AI:
            if not self._is_local_turn():
                return
            self.game.pass_turn()
            self._update_ui()
            if self.game.phase == Phase.SCORING:
                self.statusBar().showMessage("双方停一手，进入数子", 5000)
            else:
                self._start_ai()
        elif self.mode == MODE_HOST and self.server is not None:
            ok, err = self.server.host_pass()
            if not ok:
                self.statusBar().showMessage(err, 3000)
        elif self.mode == MODE_GUEST and self.client is not None:
            if self._is_local_turn():
                self.client.send_pass()

    def _resign(self):
        if not self._can_resign():
            return
        answer = QMessageBox.question(
            self, "认输", "确定认输吗？",
            QMessageBox.Yes | QMessageBox.No, QMessageBox.No,
        )
        if answer != QMessageBox.Yes:
            return
        if self.mode == MODE_AI:
            self.game.resign(self.human_color)
            self._update_ui()
            self._show_result_from_game()
        elif self.mode == MODE_HOST and self.server is not None:
            self.server.host_resign()
        elif self.mode == MODE_GUEST and self.client is not None:
            self.client.send_resign()

    def _undo(self):
        if not self._can_undo():
            return
        self._cancel_ai()
        count = 1
        if (
            len(self.game.history) >= 2
            and self.game.history[-1][0] == self.ai_color
        ):
            count = 2
        if self.game.undo(count):
            self.result_shown = False
            self._update_ui()
            if self.game.turn == self.ai_color:
                self._start_ai()

    def _confirm_score(self):
        if not self._can_mark_dead():
            return
        if self.mode == MODE_AI:
            self.game.confirm_score()
            self._update_ui()
            self._show_result_from_game()
        elif self.mode == MODE_HOST and self.server is not None:
            self.server.host_confirm_score()

    def _start_ai(self):
        if self.mode != MODE_AI or self.game.phase != Phase.PLAYING:
            return
        if self.game.turn != self.ai_color or self.ai_engine is None:
            return
        if self.ai_worker is not None:
            return
        self.ai_generation += 1
        generation = self.ai_generation
        worker = AIWorker(self.ai_engine, self.game.snapshot(), self)
        worker.move_ready.connect(
            lambda move, value=generation: self._on_ai_move(move, value)
        )
        worker.finished.connect(worker.deleteLater)
        self.ai_worker = worker
        self.statusBar().showMessage("AI 思考中…")
        self._update_ui()
        worker.start()

    def _on_ai_move(self, move, generation):
        if generation != self.ai_generation or self.mode != MODE_AI:
            return
        self.ai_worker = None
        if self.game.phase != Phase.PLAYING or self.game.turn != self.ai_color:
            self._update_ui()
            return
        if move is None:
            self.game.pass_turn()
        else:
            ok, _err = self.game.play(move[0], move[1])
            if not ok:
                self.game.pass_turn()
        self._update_ui()
        if self.game.phase == Phase.SCORING:
            self.statusBar().showMessage("双方停一手，进入数子", 5000)

    def _send_chat(self, text):
        if not text:
            return
        if self.mode == MODE_HOST and self.server is not None:
            ok, err = self.server.host_chat(text)
            if not ok:
                self.statusBar().showMessage(err, 3000)
        elif self.mode == MODE_GUEST and self.client is not None:
            self.client.send_chat(text)

    def _on_chat(self, message):
        name = str(message.get("name", ""))
        text = str(message.get("text", ""))
        stamp = message.get("timestamp")
        if isinstance(stamp, (int, float)):
            time_text = datetime.fromtimestamp(stamp / 1000.0).strftime("%H:%M")
        else:
            time_text = datetime.now().strftime("%H:%M")
        self.chat.add_message(
            name, text, time_text, is_self=(name == self.local_name)
        )

    def _show_result_from_game(self):
        self._show_result(self.game.score, self.game.result_reason)

    def _show_result(self, score, reason):
        if self.result_shown:
            return
        self.result_shown = True
        if score:
            text = score_text(score)
            text += (
                "\n\n黑：{} 点（{} 子 + {} 空）"
                "\n白：{} 点 + 贴目 {:.1f} = {:.1f} 点"
                "\n结束原因：{}"
            ).format(
                score["black_area"], score["black_stones"],
                score["black_territory"], score["white_area"],
                score["komi"], score["white_total"], reason or "数子",
            )
        elif self.game.winner:
            side = "黑" if self.game.winner == "black" else "白"
            text = "{}方获胜（{}）".format(side, reason or "对局结束")
        else:
            text = "对局结束（{}）".format(reason or "未知")
        allow_replay = self.mode == MODE_AI and self.last_ai_config is not None
        dialog = ResultDialog(text, allow_replay, self)
        dialog.exec_()
        if dialog.replay and self.last_ai_config is not None:
            size, difficulty, color = self.last_ai_config
            self._start_ai_game(size, difficulty, color)

    def _show_error(self, text):
        self.statusBar().showMessage(str(text), 5000)

    def _show_rules(self):
        QMessageBox.information(
            self, "规则说明",
            "采用中国规则数子，贴目 7.5。\n"
            "禁止自杀，禁止重复局面（位置超级劫）。\n"
            "双方连续停一手后进入数子阶段，可点击棋组标记死子，确认后计算胜负。\n"
            "局域网对战由主机校验落子；聊天消息仅当前房间可见。",
        )

    def _show_about(self):
        QMessageBox.information(
            self, "关于",
            "围棋\n桌面围棋软件，支持人机对战与局域网对战。\n"
            "PyQt5 / Python 3.7",
        )

    def closeEvent(self, event):
        self._stop_session()
        super().closeEvent(event)
```
